    <button id="spinBtn" onclick="spin()">QUAY (100đ)</button>

    <script>
        const user = "Player_" + Math.floor(Math.random()*1000);
        
        async function load() {
            const res = await fetch(`/api/init/${user}`);
            const data = await res.json();
            document.getElementById('jp-val').innerText = data.jackpot.toLocaleString();
            document.getElementById('bal-val').innerText = "Ví: " + data.balance.toLocaleString() + " xu";
        }
        load();

        async function spin() {
            const btn = document.getElementById('spinBtn');
            btn.disabled = true;
            
            const res = await fetch('/api/spin', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({ username: user })
            });
            const data = await res.json();

            let count = 0;
            const anim = setInterval(() => {
                document.querySelectorAll('.reel').forEach(el => el.innerText = ['🍎','🍋','💎'][Math.floor(Math.random()*3)]);
                count++;
                if(count > 12) {
                    clearInterval(anim);
                    document.getElementById('r1').innerText = data.reels[0];
                    document.getElementById('r2').innerText = data.reels[1];
                    document.getElementById('r3').innerText = data.reels[2];
                    document.getElementById('jp-val').innerText = data.jackpot.toLocaleString();
                    document.getElementById('bal-val').innerText = "Ví: " + data.balance.toLocaleString() + " xu";
                    if(data.isJackpot) alert("🎉 NỔ HŨ KHỔNG LỒ! 🎉");
                    btn.disabled = false;
                }
            }, 80);
        }
    </script>
