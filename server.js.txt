const app = express();

app.use(express.json());
app.use(cors());
app.use(express.static('public'));

// 2. MÔ HÌNH NGƯỜI CHƠI
const User = mongoose.model('User', new mongoose.Schema({
    username: String,
    balance: { type: Number, default: 10000 }
}));

// 3. CẤU HÌNH TỶ LỆ THẮNG (TỔNG 100%)
const SYMBOLS = [
    { name: '🍎', weight: 50 }, // Dễ ra nhất
    { name: '🍋', weight: 25 },
    { name: '🍒', weight: 15 },
    { name: '🔔', weight: 8 },
    { name: '💎', weight: 2 }  // Khó ra nhất (Dùng để nổ hũ)
];

let globalJackpot = 50000;

function getSymbol() {
    let rand = Math.random() * 100;
    let sum = 0;
    for (let s of SYMBOLS) {
        sum += s.weight;
        if (rand <= sum) return s.name;
    }
    return SYMBOLS[0].name;
}

// 4. CÁC API CHÍNH
app.get('/api/init/:user', async (req, res) => {
    let user = await User.findOne({ username: req.params.user });
    if (!user) user = await User.create({ username: req.params.user });
    res.json({ balance: user.balance, jackpot: globalJackpot });
});

app.post('/api/spin', async (req, res) => {
    const { username } = req.body;
    let user = await User.findOne({ username });
    if (user.balance < 100) return res.status(400).json({ error: "Hết tiền!" });

    user.balance -= 100; // Phí quay
    globalJackpot += 20; // Trích hũ

    let r = [getSymbol(), getSymbol(), getSymbol()];
    let win = 0;
    let isJackpot = false;

    if (r[0] === r[1] && r[1] === r[2]) {
        if (r[0] === '💎') {
            win = globalJackpot;
            globalJackpot = 50000;
            isJackpot = true;
        } else {
            win = 1000; // Thắng thường
        }
    }

    user.balance += win;
    await user.save();
    res.json({ reels: r, balance: user.balance, win, jackpot: globalJackpot, isJackpot });
});

app.listen(3000, () => console.log("Server is running on port 3000"));
